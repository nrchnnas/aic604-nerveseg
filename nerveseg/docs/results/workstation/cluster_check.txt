"""Read-only check: how much do near-duplicate clusters drive the fold-0 validation and test results?

Why: images in one near-duplicate cluster are nearly the same scan, so they are not independent, and a bootstrap
that resamples images treats them as if they were. This prints per-cluster mean Dice and, for each paired
difference, the 95% CI when resampling images and when resampling whole clusters. Reads the per-image CSVs that
run.py wrote; trains nothing and decides nothing (the validation rule in PLAN.md stands).
"""
import csv
from collections import Counter, defaultdict
from pathlib import Path

import numpy as np

ROOT = Path(r"C:\Users\nirac\aic604-nerveseg\nerveseg")
PRED = ROOT / "nnunet" / "predictions"
DS = {501: "Dataset501_NerveRequired", 502: "Dataset502_NerveGrouped"}
W0 = "nnUNetTrainer_250epochs"
BB = "nnUNetTrainer_250epochs__nnUNetResEncUNetMPlans"
AA = "nnUNetTrainerUSAug_250epochs"
RUNS = {501: {"W0": W0, "B": BB, "A": AA}, 502: {"W2": W0, "A2": AA}}
# (label, dataset, run a, suffix a, run b, suffix b): the difference is a minus b.
DIFFS = [
    ("B - W0, flips on", 501, BB, "", W0, ""),
    ("A - W0, flips on", 501, AA, "", W0, ""),
    ("A - W0, flips off", 501, AA, "__notta", W0, "__notta"),
    ("A flips off - on", 501, AA, "__notta", AA, ""),
    ("W0 flips off - on", 501, W0, "__notta", W0, ""),
    ("B flips off - on", 501, BB, "__notta", BB, ""),
    ("B clean-up - none", 501, BB, "__lcc", BB, ""),
    ("A2 - W2, flips on", 502, AA, "", W0, ""),
    ("A2 - W2, flips off", 502, AA, "__notta", W0, "__notta"),
    ("W2 flips off - on", 502, W0, "__notta", W0, ""),
    ("A2 flips off - on", 502, AA, "__notta", AA, ""),
    ("W2 clean-up - none", 502, W0, "__lcc", W0, ""),
]
DRAWS = 10000


def meta() -> dict[str, dict[str, str]]:
    with open(ROOT / "data" / "meta.csv", encoding="utf-8") as f:
        return {f"nerve_{int(r['id']):03d}": r for r in csv.DictReader(f)}


def load(ds: int, run: str, val: bool, suffix: str) -> dict[str, float] | None:
    p = PRED / f"{DS[ds]}__{run}__fold0{'__val' if val else ''}{suffix}_dice.csv"
    if not p.exists():
        return None
    with open(p, encoding="utf-8") as f:
        return {r["case"]: float(r["dice"]) for r in csv.DictReader(f)}


def ci_images(d: np.ndarray, rng: np.random.Generator) -> tuple[float, float]:
    idx = rng.integers(0, len(d), size=(DRAWS, len(d)))
    return tuple(np.percentile(d[idx].mean(axis=1), [2.5, 97.5]))


def ci_clusters(d: np.ndarray, cl: list[str], rng: np.random.Generator) -> tuple[float, float]:
    groups = defaultdict(list)
    for v, c in zip(d, cl):
        groups[c].append(v)
    arrs = [np.asarray(v) for v in groups.values()]
    picks = rng.integers(0, len(arrs), size=(DRAWS, len(arrs)))
    means = [np.concatenate([arrs[i] for i in row]).mean() for row in picks]
    return tuple(np.percentile(means, [2.5, 97.5]))


def main() -> None:
    m = meta()
    rng = np.random.default_rng(0)
    for ds, runs in RUNS.items():
        for val, name in ((True, "validation"), (False, "test")):
            scores = {k: load(ds, v, val, "") for k, v in runs.items()}
            scores = {k: v for k, v in scores.items() if v}
            if not scores:
                continue
            cases = list(next(iter(scores.values())))
            count = Counter(m[c]["cluster"] for c in cases)
            big = count.most_common(1)[0][0]
            print(f"=== {DS[ds]} {name}: {len(cases)} images, {len(count)} clusters, flips on, final checkpoint")
            print("cluster    n  " + "  ".join(f"{k:>6s}" for k in scores))
            for c, n in count.most_common():
                row = [np.mean([s[x] for x in cases if m[x]["cluster"] == c]) for s in scores.values()]
                print(f"{c:>7s} {n:4d}  " + "  ".join(f"{v:6.3f}" for v in row))
            for label, keep in ((f"all {len(cases)}", lambda x: True),
                                (f"cluster {big}", lambda x: m[x]["cluster"] == big),
                                ("the rest", lambda x: m[x]["cluster"] != big)):
                row = [np.mean([s[x] for x in cases if keep(x)]) for s in scores.values()]
                print(f"{label:>12s} " + "  ".join(f"{v:6.3f}" for v in row))
            print()
    print("=== paired differences: mean, 95% CI resampling images, 95% CI resampling whole clusters")
    for val, name in ((True, "validation"), (False, "test")):
        print(f"--- {name}")
        for label, ds, a, sa, b, sb in DIFFS:
            x, y = load(ds, a, val, sa), load(ds, b, val, sb)
            if not x or not y:
                continue
            cases = sorted(set(x) & set(y))
            d = np.array([x[c] - y[c] for c in cases])
            cl = [m[c]["cluster"] for c in cases]
            big = Counter(cl).most_common(1)[0][0]
            rest = d[[c != big for c in cl]].mean()
            li, hi = ci_images(d, rng)
            lc, hc = ci_clusters(d, cl, rng)
            print(f"{DS[ds][:10]} {label:20s} {d.mean():+.4f}  images {li:+.3f} to {hi:+.3f}  "
                  f"clusters {lc:+.3f} to {hc:+.3f}  without cluster {big}: {rest:+.4f}")
    print()
    print("=== why the two test means cannot be compared directly")
    x, y = load(501, W0, False, ""), load(502, W0, False, "")
    for name, s, col in (("W0, 501 test", x, W0), ("W2, 502 test", y, W0)):
        ds = 501 if s is x else 502
        with open(PRED / f"{DS[ds]}__{col}__fold0_dice.csv", encoding="utf-8") as f:
            rows = list(csv.DictReader(f))
        cases = list(s)
        cl = [m[c]["cluster"] for c in cases]
        d = np.array([s[c] for c in cases])
        per = defaultdict(list)
        for v, c in zip(d, cl):
            per[c].append(v)
        li, hi = ci_images(d, rng)
        lc, hc = ci_clusters(d, cl, rng)
        print(f"{name}: mean {d.mean():.4f}, each cluster counted once {np.mean([np.mean(v) for v in per.values()]):.4f}, "
              f"clusters {len(per)}, largest {Counter(cl).most_common(1)[0]}, CI images {li:.3f} to {hi:.3f}, "
              f"CI clusters {lc:.3f} to {hc:.3f}, nerve sizes {dict(Counter(r['size_group'] for r in rows))}, "
              f"median area {np.median([float(r['gt_px']) for r in rows]):.0f} px")
    print()
    print("=== the same images: the 501 model (may have near-duplicates in training) against the 502 model (has none).")
    print("Images are 501's test images that the 502 model did not train on: its test or validation images. Both 502")
    print("models used default inference, so those validation images chose nothing here. Added after seeing W2's")
    print("result, not a fixed rule.")
    with open(PRED / f"{DS[501]}__{W0}__fold0_dice.csv", encoding="utf-8") as f:
        near = {r["case"]: float(r["nearest_train_r"]) for r in csv.DictReader(f)}
    for run, n1, n2 in ((W0, "W0", "W2"), (AA, "A", "A2")):
        x1, t2, v2 = load(501, run, False, ""), load(502, run, False, ""), load(502, run, True, "")
        if not (x1 and t2 and v2):
            continue
        held = {**v2, **t2}
        shared = sorted(set(x1) & set(held))
        groups = ((f"{n1} has a near-duplicate (r > 0.95)", [c for c in shared if near[c] > 0.95]),
                  (f"{n1} has none (r <= 0.95)", [c for c in shared if near[c] <= 0.95]),
                  ("all", shared))
        print(f"--- {n1} against {n2}")
        for label, cases in groups:
            d = np.array([x1[c] - held[c] for c in cases])
            cl = [m[c]["cluster"] for c in cases]
            li, hi = ci_images(d, rng)
            lc, hc = ci_clusters(d, cl, rng)
            print(f"{label}: n={len(cases)}, {len(set(cl))} clusters, {n1} {np.mean([x1[c] for c in cases]):.4f}, "
                  f"{n2} {np.mean([held[c] for c in cases]):.4f}, {n1} - {n2} mean {d.mean():+.4f}, "
                  f"median {np.median(d):+.4f}, CI images {li:+.3f} to {hi:+.3f}, CI clusters {lc:+.3f} to {hc:+.3f}, "
                  f"{n1} better {(d > 0).sum()}, {n2} better {(d < 0).sum()}")
        for c in shared:
            print(f"  {c} cluster {m[c]['cluster']:>3s}  nearest r {near[c]:.3f}  {n1} {x1[c]:.3f}  {n2} {held[c]:.3f}")


if __name__ == "__main__":
    main()

"""Read-only check: does this machine, today, give the same masks as the ones saved earlier?

Why: fold 3 crashed on 6 Nov with an illegal instruction, and the CPU may be unstable. If a saved model predicted
again gives the same masks, the checkpoint and the saved predictions agree. This writes nothing into the project.
It covers W0 on the 60 test images, then fold 3 (the one that crashed) and fold 4 (lowest score, trained straight
after the crash) on their own validation images. Same settings as nnU-Net's own: final checkpoint, flip averaging on.
"""
import json
import os
from pathlib import Path

import numpy as np
import torch

N = Path(r"C:\Users\nirac\aic604-nerveseg\nerveseg\nnunet")
os.environ.update(nnUNet_raw=str(N / "nnUNet_raw"), nnUNet_preprocessed=str(N / "nnUNet_preprocessed"),
                  nnUNet_results=str(N / "nnUNet_results"), nnUNet_extTrainer=str(N / "trainers"))

from nnunetv2.imageio.natural_image_reader_writer import NaturalImage2DIO  # noqa: E402
from nnunetv2.inference.predict_from_raw_data import nnUNetPredictor  # noqa: E402

DS = "Dataset501_NerveRequired"
RAW = N / "nnUNet_raw" / DS
MODEL = N / "nnUNet_results" / DS / "nnUNetTrainer_250epochs__nnUNetPlans__2d"
PRED = N / "predictions" / f"{DS}__nnUNetTrainer_250epochs__fold0"


def dice(a: np.ndarray, b: np.ndarray) -> float:
    s = a.sum() + b.sum()
    return 1.0 if s == 0 else float(2 * np.logical_and(a, b).sum() / s)


def check(fold: int, name: str, cases: list[str], img_dir: Path, gt_dir: Path, saved_dir: Path, io) -> None:
    p = nnUNetPredictor(tile_step_size=0.5, use_gaussian=True, use_mirroring=True, perform_everything_on_device=True,
                        device=torch.device("cuda"), verbose=False, verbose_preprocessing=False, allow_tqdm=False)
    p.initialize_from_trained_model_folder(str(MODEL), use_folds=(fold,), checkpoint_name="checkpoint_final.pth")
    print(f"=== fold {fold}, {name}: {len(cases)} images, saved masks in {saved_dir}", flush=True)
    diffs, overlaps, new_d, old_d = [], [], [], []
    for case in cases:
        img, props = io.read_images([str(img_dir / f"{case}_0000.png")])
        gt = io.read_seg(str(gt_dir / f"{case}.png"))[0][0] > 0
        saved = io.read_seg(str(saved_dir / f"{case}.png"))[0][0] > 0
        new = p.predict_single_npy_array(img, props, None, None, False)[0] > 0
        diffs.append(int(np.logical_xor(saved, new).sum()))
        overlaps.append(dice(saved, new))
        new_d.append(dice(new, gt))
        old_d.append(dice(saved, gt))
    diffs = np.array(diffs)
    print(f"identical masks: {(diffs == 0).sum()} of {len(cases)}; pixels differing: total {diffs.sum()}, "
          f"largest in one image {diffs.max()}; lowest saved-vs-new overlap {min(overlaps):.5f}")
    print(f"mean Dice saved {np.mean(old_d):.6f}, predicted today {np.mean(new_d):.6f}, "
          f"difference {np.mean(new_d) - np.mean(old_d):+.6f}; "
          f"largest change in one image {np.abs(np.array(new_d) - np.array(old_d)).max():.5f}", flush=True)


def main() -> None:
    io = NaturalImage2DIO()
    splits = json.load(open(N / "nnUNet_preprocessed" / DS / "splits_final.json"))
    test = sorted(f.stem for f in (RAW / "labelsTs").glob("*.png"))
    check(0, "test", test, RAW / "imagesTs", RAW / "labelsTs", PRED, io)
    for fold in (3, 4):
        check(fold, "own validation", splits[fold]["val"], RAW / "imagesTr", RAW / "labelsTr",
              MODEL / f"fold_{fold}" / "validation", io)
    print("done", flush=True)


if __name__ == "__main__":
    main()

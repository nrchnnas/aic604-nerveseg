=== script
"""Read-only check of option C: why do the five folds score so differently, and how firm is the ensemble's gain?

Why: every fold holds out whole near-duplicate clusters, so fold-to-fold differences may come from which scans are
held out. A resampling of images treats the 240 images as independent, and they are not, so this also resamples whole
clusters. Nerve position is measured because W2's zero-Dice images had the nerve far from where training put it.
Reads the masks and CSVs that run.py wrote; trains nothing and decides nothing.
"""
import csv
import json
import re
from collections import Counter, defaultdict
from pathlib import Path

import numpy as np
from PIL import Image
from scipy.stats import spearmanr

ROOT = Path(r"C:\Users\nirac\aic604-nerveseg\nerveseg")
PRED = ROOT / "nnunet" / "predictions"
DS = "Dataset501_NerveRequired"
TR = "nnUNetTrainer_250epochs"
CV = PRED / f"{DS}__{TR}__cv_folds01234"
SPLITS = ROOT / "nnunet" / "nnUNet_preprocessed" / DS / "splits_final.json"
RESULTS = ROOT / "nnunet" / "nnUNet_results" / DS / f"{TR}__nnUNetPlans__2d"
DRAWS = 10000


def read(path: Path) -> list[dict[str, str]]:
    with open(path, encoding="utf-8") as f:
        return list(csv.DictReader(f))


def center(path: Path) -> tuple[float, float] | None:
    ys, xs = np.nonzero(np.asarray(Image.open(path)) > 0)
    return None if len(xs) == 0 else (float(xs.mean()), float(ys.mean()))


def ci_images(d: np.ndarray, rng: np.random.Generator) -> tuple[float, float]:
    idx = rng.integers(0, len(d), size=(DRAWS, len(d)))
    return tuple(np.percentile(d[idx].mean(axis=1), [2.5, 97.5]))


def ci_clusters(d: np.ndarray, cl: list[str], rng: np.random.Generator) -> tuple[float, float]:
    groups = defaultdict(list)
    for v, c in zip(d, cl):
        groups[c].append(v)
    arrs = [np.asarray(v) for v in groups.values()]
    picks = rng.integers(0, len(arrs), size=(DRAWS, len(arrs)))
    means = [np.concatenate([arrs[i] for i in row]).mean() for row in picks]
    return tuple(np.percentile(means, [2.5, 97.5]))


def training_table() -> None:
    print("=== training: last 10 epochs, each fold")
    print("fold  epochs  train loss  val loss    gap  lowest val loss (epoch)  EMA pseudo Dice highest (epoch), last")
    for f in range(5):
        tr, va, ema = [], [], []
        for log in sorted((RESULTS / f"fold_{f}").glob("training_log_*.txt")):
            for line in open(log, encoding="utf-8"):
                if m := re.search(r"train_loss (-?[\d.]+)", line):
                    tr.append(float(m.group(1)))
                if m := re.search(r"val_loss (-?[\d.]+)", line):
                    va.append(float(m.group(1)))
        t, v = np.mean(tr[-10:]), np.mean(va[-10:])
        import torch
        c = torch.load(RESULTS / f"fold_{f}" / "checkpoint_final.pth", map_location="cpu", weights_only=False)
        ema = c["logging"]["ema_fg_dice"]
        print(f"{f:4d}  {len(tr):6d}  {t:10.3f}  {v:8.3f}  {v - t:5.3f}  {min(va):8.3f} ({int(np.argmin(va)) + 1:3d})"
              f"             {max(ema):.4f} ({int(np.argmax(ema)) + 1:3d}), {ema[-1]:.4f}")
    print()


def main() -> None:
    rng = np.random.default_rng(0)
    meta = {f"nerve_{int(r['id']):03d}": r for r in read(ROOT / "data" / "meta.csv")}
    splits = json.load(open(SPLITS))
    fold_of = {c: f for f, s in enumerate(splits) for c in s["val"]}
    rows = {r["case"]: r for r in read(Path(f"{CV}_dice.csv"))}
    cases = sorted(rows)
    dice = {c: float(rows[c]["dice"]) for c in cases}
    cluster = {c: meta[c]["cluster"] for c in cases}
    gt = {c: center(CV / "labels" / f"{c}.png") for c in cases}
    pr = {c: center(CV / f"{c}.png") for c in cases}

    training_table()

    print("=== cross-validation over the 240 training images")
    d = np.array([dice[c] for c in cases])
    cl = [cluster[c] for c in cases]
    per = defaultdict(list)
    for c in cases:
        per[cluster[c]].append(dice[c])
    means = [np.mean(v) for v in per.values()]
    print(f"mean {d.mean():.4f}, median {np.median(d):.4f}, zero Dice {(d == 0).sum()} of {len(d)}")
    print("95% CI resampling images {:.3f} to {:.3f}; resampling whole clusters {:.3f} to {:.3f}".format(
        *ci_images(d, rng), *ci_clusters(d, cl, rng)))
    print(f"{len(per)} clusters; each counted once: mean {np.mean(means):.4f}, median {np.median(means):.4f}")
    fm = [np.mean([dice[c] for c in cases if fold_of[c] == f]) for f in range(5)]
    print(f"mean of the five fold means {np.mean(fm):.4f}, SD {np.std(fm, ddof=1):.4f}, range {min(fm):.4f} to {max(fm):.4f}")
    print()

    print("=== each fold, with nerve position relative to the x position in that model's training images")
    print("fold    n  clusters   mean  median  zeros  each cluster once  found poor elsewhere empty   mean |x z|")
    zs = {}
    for f in range(5):
        val = [c for c in cases if fold_of[c] == f]
        train_x = np.array([gt[c][0] for c in cases if fold_of[c] != f])
        mu, sd = train_x.mean(), train_x.std(ddof=1)
        for c in val:
            zs[c] = (gt[c][0] - mu) / sd
        v = np.array([dice[c] for c in val])
        pc = defaultdict(list)
        for c in val:
            pc[cluster[c]].append(dice[c])
        fail = Counter(rows[c]["failure"] for c in val)
        print(f"{f:4d} {len(val):4d} {len(pc):9d} {v.mean():6.3f} {np.median(v):7.3f} {(v == 0).sum():6d} "
              f"{np.mean([np.mean(x) for x in pc.values()]):18.3f} {fail.get('found', 0):5d} {fail.get('poor', 0):4d} "
              f"{fail.get('missed_elsewhere', 0):9d} {fail.get('missed_empty', 0):5d} "
              f"{np.mean([abs(zs[c]) for c in val]):11.2f}   (training x mean {mu:.0f}, SD {sd:.0f})")
    print()

    print("=== clusters within each fold (n >= 2), lowest first; x is the mean horizontal centre in pixels")
    print("fold cluster    n   mean  zeros  true x  predicted x  x z")
    for f in range(5):
        val = [c for c in cases if fold_of[c] == f]
        by = defaultdict(list)
        for c in val:
            by[cluster[c]].append(c)
        for k, group in sorted(by.items(), key=lambda kv: np.mean([dice[c] for c in kv[1]])):
            if len(group) < 2:
                continue
            px = [pr[c][0] for c in group if pr[c]]
            print(f"{f:4d} {k:>7s} {len(group):4d} {np.mean([dice[c] for c in group]):6.3f} "
                  f"{sum(dice[c] == 0 for c in group):6d} {np.mean([gt[c][0] for c in group]):7.0f} "
                  f"{np.mean(px) if px else float('nan'):12.0f} {np.mean([zs[c] for c in group]):+5.2f}")
        rest = [c for g in by.values() if len(g) < 2 for c in g]
        if rest:
            print(f"{f:4d} singles {len(rest):4d} {np.mean([dice[c] for c in rest]):6.3f} "
                  f"{sum(dice[c] == 0 for c in rest):6d}")
    print()

    print("=== does Dice fall as the nerve sits further from the training position? (240 images)")
    az = np.array([abs(zs[c]) for c in cases])
    rho, p = spearmanr(az, d)
    print(f"Spearman rho, |x z| vs Dice: {rho:.3f} (p = {p:.2g}; images in a cluster are not independent, so p is too small)")
    pc = {k: (np.mean([abs(zs[c]) for c in cases if cluster[c] == k]), np.mean(v)) for k, v in per.items()}
    rho, p = spearmanr([v[0] for v in pc.values()], [v[1] for v in pc.values()])
    print(f"same, one point per cluster: rho {rho:.3f} (p = {p:.2g}, {len(pc)} clusters)")
    for lo, hi in ((0, 1), (1, 2), (2, 99)):
        keep = (az >= lo) & (az < hi)
        print(f"  |x z| {lo} to {hi if hi < 99 else 'more'}: n={keep.sum():3d}, mean Dice {d[keep].mean():.3f}, "
              f"zero Dice {(d[keep] == 0).sum()}")
    far = [c for c in cases if dice[c] == 0 and pr[c]]
    opposite = sum(np.sign(pr[c][0] - gt[c][0]) * np.sign(zs[c]) < 0 for c in far)
    print(f"zero-Dice images with a prediction: {len(far)}; predicted centre lies from the true centre back toward the "
          f"training mean in {opposite} of them; mean gap {np.mean([abs(pr[c][0] - gt[c][0]) for c in far]):.0f} px")
    print()

    print("=== test images: ensemble of five folds against the single fold-0 model (W0), flips on, final checkpoint")
    ens = {r["case"]: r for r in read(PRED / f"{DS}__{TR}__folds01234_dice.csv")}
    one = {r["case"]: r for r in read(PRED / f"{DS}__{TR}__fold0_dice.csv")}
    tc = sorted(ens)
    near = {c: float(one[c]["nearest_train_r"]) for c in tc}
    tcl = [meta[c]["cluster"] for c in tc]
    diff = np.array([float(ens[c]["dice"]) - float(one[c]["dice"]) for c in tc])
    print(f"ensemble {np.mean([float(ens[c]['dice']) for c in tc]):.4f}, single {np.mean([float(one[c]['dice']) for c in tc]):.4f}, "
          f"difference {diff.mean():+.4f}, median {np.median(diff):+.4f}")
    print("95% CI resampling images {:+.3f} to {:+.3f}; resampling whole clusters {:+.3f} to {:+.3f} ({} clusters)".format(
        *ci_images(diff, rng), *ci_clusters(diff, tcl, rng), len(set(tcl))))
    print(f"better {(diff > 0).sum()}, worse {(diff < 0).sum()}")
    order = np.argsort(-diff)
    for i in order[:3]:
        print(f"  largest gain: {tc[i]} cluster {tcl[i]}, single {float(one[tc[i]]['dice']):.3f}, ensemble {float(ens[tc[i]]['dice']):.3f}")
    print(f"  largest loss: {tc[order[-1]]} cluster {tcl[order[-1]]}, single {float(one[tc[order[-1]]]['dice']):.3f}, ensemble {float(ens[tc[order[-1]]]['dice']):.3f}")
    print(f"  without the largest gain: {np.delete(diff, order[0]).mean():+.4f}; without the largest three: {np.delete(diff, order[:3]).mean():+.4f}")
    for label, keep in (("with a near-duplicate among the 240 training images (r > 0.95)", [near[c] > 0.95 for c in tc]),
                        ("without one", [near[c] <= 0.95 for c in tc])):
        k = np.array(keep)
        dk = diff[k]
        kc = [x for x, y in zip(tcl, keep) if y]
        print(f"{label}: n={k.sum()}, difference {dk.mean():+.4f}, CI images {ci_images(dk, rng)[0]:+.3f} to {ci_images(dk, rng)[1]:+.3f}, "
              f"CI clusters {ci_clusters(dk, kc, rng)[0]:+.3f} to {ci_clusters(dk, kc, rng)[1]:+.3f}, {len(set(kc))} clusters")
    print(f"regions: images with more than one predicted region, single {sum(int(one[c]['n_pred_regions']) > 1 for c in tc)}, "
          f"ensemble {sum(int(ens[c]['n_pred_regions']) > 1 for c in tc)}")
    print()

    print("=== why cross-validation is lower than test")
    print("43 of 60 test images are near-duplicates of one of the 240 training images; no held-out image in cross-validation has one.")
    vals = [dice[c] for c in cases]
    print(f"CV (no cluster shared): {np.mean(vals):.4f}. W0 test, near-duplicate: "
          f"{np.mean([float(one[c]['dice']) for c in tc if near[c] > 0.95]):.4f}; W0 test, none: "
          f"{np.mean([float(one[c]['dice']) for c in tc if near[c] <= 0.95]):.4f}")
    same = [c for c in tc if near[c] <= 0.95]
    tr_cl = Counter(meta[c]["cluster"] for c in splits[0]["train"])
    print(f"the {len(same)} W0 test images without a near-duplicate: {sum(tr_cl.get(meta[c]['cluster'], 0) > 0 for c in same)} "
          f"still share a cluster with fold 0's training images: clusters chain images at r > 0.80, while the "
          f"near-duplicate flag uses r > 0.95, so they are not fully held out either")
    print()

    print("=== how concentrated are the failures? (picked after seeing the results, to describe them, not to drop images)")
    zero = Counter(cluster[c] for c in cases if dice[c] == 0)
    print(f"zero-Dice images by cluster: {zero.most_common()}")
    worst = ["12", "14", "2"]
    keep = [c for c in cases if cluster[c] not in worst]
    print(f"clusters {worst} hold {sum(cluster[c] in worst for c in cases)} of 240 images and "
          f"{sum(zero[k] for k in worst)} of {sum(zero.values())} zero-Dice images; the other {len(keep)} images: "
          f"mean {np.mean([dice[c] for c in keep]):.4f}, median {np.median([dice[c] for c in keep]):.4f}")
    for f in (1, 4):
        v = [dice[c] for c in cases if fold_of[c] == f and cluster[c] not in worst]
        print(f"  fold {f} without them: n={len(v)}, mean {np.mean(v):.4f}")
    print()

    print("=== image field width: columns of the image that hold ultrasound, not black margin")
    raw = ROOT / "nnunet" / "nnUNet_raw" / DS / "imagesTr"
    width = {c: int((np.asarray(Image.open(raw / f"{c}_0000.png").convert("L")).mean(axis=0) > 5).sum()) for c in cases}
    w = np.array([width[c] for c in cases])
